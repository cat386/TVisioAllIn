cmake_minimum_required(VERSION 3.16)
if (CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(tvstudio LANGUAGES C CXX)
endif()
get_filename_component(TVSTUDIO_FRAMEWORK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if (NOT TARGET tvision)
    set(TV_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(TV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${TVSTUDIO_FRAMEWORK_ROOT}" "${CMAKE_BINARY_DIR}/tvision")
endif()
find_package(Threads REQUIRED)
add_library(tvstudio-model STATIC src/model.cpp src/codec.cpp)
target_include_directories(tvstudio-model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(tvstudio-model PUBLIC cxx_std_17)
target_link_libraries(tvstudio-model PUBLIC tvision)
if (MSVC)
    target_compile_options(tvstudio-model PUBLIC /utf-8)
endif()
add_library(tvstudio-runtime STATIC src/runtime.cpp)
target_link_libraries(tvstudio-runtime PUBLIC tvstudio-model)
option(TVSTUDIO_BUILD_IDE "Build TV Studio C++ executable" ON)
if (TVSTUDIO_BUILD_IDE OR TV_BUILD_TESTS)
    add_library(tvstudio-core STATIC src/codegen.cpp src/process.cpp src/lsp.cpp)
    target_link_libraries(tvstudio-core PUBLIC tvstudio-runtime Threads::Threads)
    if (WIN32)
        target_link_libraries(tvstudio-core PRIVATE shell32)
    endif()
endif()
if (TVSTUDIO_BUILD_IDE)
    add_executable(tvstudio src/studio.cpp)
    target_link_libraries(tvstudio PRIVATE tvstudio-core)
    target_compile_definitions(tvstudio PRIVATE TVSTUDIO_SOURCE_ROOT="${TVSTUDIO_FRAMEWORK_ROOT}")
    tv_set_output_dir(tvstudio)
endif()
if (TV_BUILD_TESTS)
    add_executable(tvstudio-clangd-test tests/clangd.cpp)
    target_link_libraries(tvstudio-clangd-test PRIVATE tvstudio-core)
    tv_set_output_dir(tvstudio-clangd-test)
    add_executable(tvstudio-tests tests/model.test.cpp "${TVSTUDIO_FRAMEWORK_ROOT}/test/tvtest.cpp")
    target_include_directories(tvstudio-tests PRIVATE "${TVSTUDIO_FRAMEWORK_ROOT}/test")
    target_link_libraries(tvstudio-tests PRIVATE tvstudio-core)
    add_executable(tvstudio-generate-fixture tests/generate.cpp)
    target_link_libraries(tvstudio-generate-fixture PRIVATE tvstudio-core)
    tv_set_output_dir(tvstudio-generate-fixture)
    add_executable(tvstudio-process-fixture tests/process.cpp)
    target_link_libraries(tvstudio-process-fixture PRIVATE tvstudio-core)
    tv_set_output_dir(tvstudio-process-fixture)
    target_compile_definitions(tvstudio-tests PRIVATE TVSTUDIO_PROCESS_FIXTURE="$<TARGET_FILE:tvstudio-process-fixture>")
    add_dependencies(tvstudio-tests tvstudio-process-fixture)
    tv_set_output_dir(tvstudio-tests)
    add_custom_command(OUTPUT tvstudio-tests-passed
        COMMAND $<TARGET_FILE:tvstudio-tests>
        COMMAND ${CMAKE_COMMAND} -E touch tvstudio-tests-passed
        DEPENDS tvstudio-tests WORKING_DIRECTORY ${TV_BINARY_DIR})
    add_custom_target(tvstudio-tests-run ALL DEPENDS tvstudio-tests-passed)
    if (TVSTUDIO_BUILD_IDE)
        add_custom_target(tvstudio-smoke ALL COMMAND $<TARGET_FILE:tvstudio> --smoke DEPENDS tvstudio)
    endif()
endif()
